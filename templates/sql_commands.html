Syllabus & Core Query Reference

1. Multi-Table JOIN & Analysis

SELECT a.name AS artist, al.title AS album, COUNT(s.song_id) AS total_tracks,
       ROUND(SUM(s.file_size) / (1024.0*1024.0), 2) AS total_album_mb
FROM Songs s
JOIN Albums al ON s.album_id = al.album_id
JOIN Artists a ON al.artist_id = a.artist_id
GROUP BY al.album_id
ORDER BY total_tracks DESC
LIMIT 10;

2. High-Resolution Audio Filter (Bit Depth >= 24)

SELECT s.title, a.name AS artist, aq.bit_depth, aq.sample_rate, aq.bitrate
FROM Songs s
JOIN Albums al ON s.album_id = al.album_id
JOIN Artists a ON al.artist_id = a.artist_id
JOIN AudioQuality aq ON s.quality_id = aq.quality_id
WHERE aq.bit_depth >= 24
ORDER BY aq.sample_rate DESC, aq.bitrate DESC
LIMIT 15;

3. Aggregations (Storage & Duration Statistics)

SELECT 
    COUNT(*) AS total_songs,
    ROUND(SUM(file_size) / (1024.0*1024.0*1024.0), 2) AS total_gb,
    ROUND(AVG(duration), 1) AS avg_duration_sec,
    MAX(file_size) / (1024*1024) AS max_file_mb
FROM Songs;

4. Schema DDL Structure

CREATE TABLE Artists (
    artist_id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL UNIQUE
);

CREATE TABLE Albums (
    album_id INTEGER PRIMARY KEY AUTOINCREMENT,
    artist_id INTEGER NOT NULL,
    title TEXT NOT NULL,
    release_year INTEGER,
    FOREIGN KEY (artist_id) REFERENCES Artists(artist_id)
);

CREATE TABLE Songs (
    song_id INTEGER PRIMARY KEY AUTOINCREMENT,
    album_id INTEGER NOT NULL,
    title TEXT NOT NULL,
    track_number INTEGER,
    duration REAL,
    file_size INTEGER,
    quality_id INTEGER,
    genre_id INTEGER,
    play_count INTEGER DEFAULT 0,
    rating REAL DEFAULT 0,
    FOREIGN KEY (album_id) REFERENCES Albums(album_id),
    FOREIGN KEY (genre_id) REFERENCES Genres(genre_id),
    FOREIGN KEY (quality_id) REFERENCES AudioQuality(quality_id)
);
